import pandas as pd
import matplotlib.pyplot as plt
import matplotlib.dates as mdates

# Function to load and clean power log files
def load_power_log(file_path):
    df = pd.read_csv(file_path, header=None, names=['Timestamp', 'Power'], sep=',', engine='python')
    df['Timestamp'] = pd.to_datetime(df['Timestamp'])
    df['Power'] = df['Power'].astype(str).str.replace(' mW', '').astype(float)
    return df

# Load both datasets
dutycycle_df = load_power_log('esp32_dutycycle_power_log.txt')
onlypower_df = load_power_log('esp32_power_log_only_power.txt')

# Plotting function
def plot_power_data(df, title):
    plt.figure(figsize=(16, 6))
    plt.bar(df['Timestamp'], df['Power'], width=0.01)
    plt.gca().xaxis.set_major_formatter(mdates.DateFormatter('%m-%d %H:%M'))
    plt.xlabel('Time')
    plt.ylabel('Power (mW)')
    plt.title(title)
    plt.tight_layout()
    plt.grid(True)
    plt.show()

# Plot each graph
plot_power_data(dutycycle_df, 'ESP32 Duty-Cycle Power Usage')
plot_power_data(onlypower_df, 'ESP32 Continuous Power Usage')
